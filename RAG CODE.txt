from transformers import pipeline
from sentence_transformers import SentenceTransformer
import torch

# Load Qwen LLM
chatbot = pipeline(
    "text-generation",
    model="Qwen/Qwen2.5-0.5B-Instruct",
    device=0
)

chatbot.model.generation_config.max_length = None

# Knowledge Base
documents = [
    """
    Our college offers undergraduate programs in Computer Science,
    Artificial Intelligence and Data Science, Electronics and Communication,
    and Mechanical Engineering.
    """,

    """
    The Computer Science department teaches programming languages,
    data structures, algorithms, databases, computer networks,
    operating systems, and artificial intelligence.
    """,

    """
    Students can use the college library to access textbooks,
    reference books, journals, and digital learning resources.
    """,

    """
    The college provides laboratories for programming, computer networks,
    electronics, and other practical learning activities.
    """,

    """
    Students can participate in technical events, workshops,
    hackathons, coding competitions, and project activities.
    """
]

print("Knowledge base created!")

# Create embedding model
embedding_model = SentenceTransformer(
    "sentence-transformers/all-MiniLM-L6-v2"
)

# Convert documents into embeddings
document_embeddings = embedding_model.encode(
    documents,
    convert_to_tensor=True
)

print("Document embeddings created!")

# Retrieval function
def retrieve_documents(query, top_k=2):

    query_embedding = embedding_model.encode(
        query,
        convert_to_tensor=True
    )

    scores = torch.nn.functional.cosine_similarity(
        query_embedding,
        document_embeddings
    )

    top_results = torch.topk(scores, k=top_k)

    retrieved_docs = [
        documents[i] for i in top_results.indices
    ]

    return retrieved_docs


# Start RAG chatbot
print("\n🤖 RAG Chatbot Started!")
print("Type 'exit' to stop.\n")

while True:

    user_input = input("You: ")

    if user_input.lower() == "exit":
        print("Bot: Goodbye!")
        break

    # Retrieve relevant information
    retrieved_docs = retrieve_documents(user_input)

    # Combine retrieved information
    context = "\n".join(retrieved_docs)

    # Create prompt
    prompt = f"""
You are a helpful college information assistant.

Answer the user's question using only the information provided below.
If the answer is not available in the knowledge base, say:
"Sorry, this information is not available in my knowledge base."

Knowledge Base:
{context}

User Question:
{user_input}
"""

    # Generate answer using Qwen
    response = chatbot(
        prompt,
        max_new_tokens=50,
        do_sample=True,
        temperature=0.7
    )

    answer = response[0]["generated_text"]

    print("Bot:", answer)
    print()
